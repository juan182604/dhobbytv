    21→// ==================== LOGIN ====================
    22→function LoginView() {
    23→  const [username, setUsername] = useState('')
    24→  const [password, setPassword] = useState('')
    25→  const [loading, setLoading] = useState(false)
    26→  const setView = useDhobbytvStore((s) => s.setView)
    27→  const setUser = useDhobbytvStore((s) => s.setUser)
    28→
    29→  const handleLogin = async () => {
    30→    if (!username || !password) return toast.error('Completa todos los campos')
    31→    setLoading(true)
    32→    try {
    33→      const res = await fetch('/api/login', {
    34→        method: 'POST',
    35→        headers: { 'Content-Type': 'application/json' },
    36→        body: JSON.stringify({ username, password }),
    37→      })
    38→      const data = await res.json()
    39→      if (data.error) return toast.error(data.error)
    40→      setUser(data.user)
    41→      if (data.user.isAdmin) {
    42→        setView('admin')
    43→      } else if (!data.user.verified) {
    44→        setView('verification')
    45→      } else {
    46→        setView('main')
    47→      }
    48→    } catch {
    49→      toast.error('Error de conexión')
    50→    } finally {
    51→      setLoading(false)
    52→    }
    53→  }
    54→
    55→  return (
    56→    <div className="min-h-screen flex items-center justify-center p-4 bg-gradient-to-br from-purple-900 via-indigo-900 to-black">
    57→      <Card className="w-full max-w-md bg-gray-900/80 border-gray-700 text-white">
    58→        <CardHeader className="text-center pb-2">
    59→          <CardTitle className="text-4xl font-black tracking-tight">
    60→            <span className="text-purple-400">dhobby</span>
    61→            <span className="text-green-400">tv</span>
    62→          </CardTitle>
    63→          <p className="text-gray-400 text-sm mt-1">Conecta con personas que comparten tus intereses</p>
    64→        </CardHeader>
    65→        <CardContent className="space-y-4">
    66→          <Input
    67→            placeholder="Usuario"
    68→            value={username}
    69→            onChange={(e) => setUsername(e.target.value)}
    70→            className="bg-gray-800 border-gray-600 text-white placeholder:text-gray-500"
    71→            onKeyDown={(e) => e.key === 'Enter' && handleLogin()}
    72→          />
    73→          <Input
    74→            type="password"
    75→            placeholder="Contraseña"
    76→            value={password}
    77→            onChange={(e) => setPassword(e.target.value)}
    78→            className="bg-gray-800 border-gray-600 text-white placeholder:text-gray-500"
    79→            onKeyDown={(e) => e.key === 'Enter' && handleLogin()}
    80→          />
    81→          <Button
    82→            onClick={handleLogin}
    83→            disabled={loading}
    84→            className="w-full bg-purple-600 hover:bg-purple-700 text-white font-bold py-6 text-lg"
    85→          >
    86→            {loading ? 'Entrando...' : 'Iniciar Sesión'}

    90→              onClick={() => setView('register')}
    91→              className="text-purple-400 hover:text-purple-300 text-sm underline"

   195→// ==================== VERIFICATION WAITING ====================
   196→function VerificationView() {
   197→  const user = useDhobbytvStore((s) => s.user)
   198→  const socketRef = useRef<Socket | null>(null)
   199→
   200→  useEffect(() => {
   201→    const socket = io('/?XTransformPort=3003', { transports: ['websocket'] })
   202→    socketRef.current = socket
   203→
   204→    socket.on('connect', () => {
   205→      socket.emit('join-verification-queue', { username: user?.username, gender: user?.gender })
   206→    })
   207→
   208→    socket.on('in-verification-queue', (data: { position: number }) => {
   209→      toast.info(`Posición en la cola: ${data.position}`)
   210→    })
   211→
   212→    socket.on('start-verification', (data: { adminSocketId: string; signal: boolean }) => {
   213→      useDhobbytvStore.getState().setVerificationAdminSocketId(data.adminSocketId)
   214→      useDhobbytvStore.getState().setView('verification-video')
   215→    })
   216→
   217→    socket.on('verification-accepted', () => {
   218→      if (user) {
   219→        useDhobbytvStore.getState().setUser({ ...user, verified: true })
   220→        fetch('/api/verify-user', {
   221→          method: 'POST',
   222→          headers: { 'Content-Type': 'application/json' },
   223→          body: JSON.stringify({ username: user.username }),
   224→        })
   225→      }
   226→      toast.success('Has sido verificado! Ya puedes usar dhobbytv')
   227→      useDhobbytvStore.getState().setView('main')
   228→      socket.disconnect()
   229→    })
   230→
   231→    socket.on('verification-rejected', () => {
   232→      if (user) {
   233→        fetch('/api/delete-user', {
   234→          method: 'POST',
   235→          headers: { 'Content-Type': 'application/json' },
   236→          body: JSON.stringify({ username: user.username }),
   237→        })
   238→      }
   239→      toast.error('Verificación rechazada. Tu cuenta ha sido eliminada.')
   240→      useDhobbytvStore.getState().setUser(null)
   241→      useDhobbytvStore.getState().setView('login')
   242→      socket.disconnect()
   243→    })
   244→
   245→    return () => { socket.disconnect() }
   246→  }, [])
   247→
   248→  return (
   249→    <div className="min-h-screen flex items-center justify-center p-4 bg-gradient-to-br from-purple-900 via-indigo-900 to-black">
   250→      <Card className="w-full max-w-md bg-gray-900/80 border-gray-700 text-white text-center">
   251→        <CardContent className="pt-8 pb-8 space-y-4">
   252→          <div className="text-6xl animate-pulse">🔒</div>
   253→          <h2 className="text-2xl font-bold">Verificación de Edad</h2>
   254→          <p className="text-gray-400">
   255→            Necesitas verificarte con un administrador para confirmar que eres mayor de edad.
   256→          </p>
   257→          <p className="text-yellow-400 text-sm">
   258→            Prepara tu documento de identificación. El admin te pedirá que lo muestres por cámara.
   259→          </p>
   260→          <div className="flex items-center justify-center gap-2 text-gray-300">
   261→            <div className="w-2 h-2 bg-green-400 rounded-full animate-pulse" />
   262→            Esperando en la cola de verificación...
   263→          </div>
   264→          <Button
   265→            variant="outline"
   266→            className="text-gray-400 border-gray-600 hover:text-white"
   267→            onClick={() => {
   268→              socketRef.current?.disconnect()
   269→              useDhobbytvStore.getState().setUser(null)
   270→              useDhobbytvStore.getState().setView('login')

   272→          >
   273→            Salir
   274→          </Button>
   275→        </CardContent>

   282→function VerificationVideoView() {

   285→  const [localStream, setLocalStream] = useState<MediaStream | null>(null)

   288→  const socketRef = useRef<Socket | null>(null)
   289→  const [message, setMessage] = useState('Conectando con administrador...')
   290→
   291→  useEffect(() => {
   292→    let mounted = true
   293→
   294→    const setup = async () => {
   295→      const stream = await navigator.mediaDevices.getUserMedia({ video: true, audio: true })
   296→      if (!mounted) { stream.getTracks().forEach((t) => t.stop()); return }
   297→      setLocalStream(stream)
   298→      if (localVideoRef.current) localVideoRef.current.srcObject = stream
   299→
   300→      const socket = io('/?XTransformPort=3003', { transports: ['websocket'] })
   301→      socketRef.current = socket
   302→
   303→      const pc = new RTCPeerConnection({
   304→        iceServers: [{ urls: 'stun:stun.l.google.com:19302' }, { urls: 'stun:stun1.l.google.com:19302' }],
   305→      })
   306→      peerRef.current = pc
   307→
   308→      stream.getTracks().forEach((track) => pc.addTrack(track, stream))
   309→
   310→      pc.onicecandidate = (e) => {
   311→        if (e.candidate && adminSocketId) {
   312→          socket.emit('webrtc-ice-candidate', { targetId: adminSocketId, candidate: e.candidate })
   313→        }
   314→      }
   315→
   316→      pc.ontrack = (e) => {
   317→        // Admin camera is off, but we still handle the connection
   318→      }
   319→
   320→      socket.on('webrtc-offer', async (data: { fromId: string; offer: RTCSessionDescriptionInit }) => {
   321→        setMessage('Conectado con administrador. Muestra tu identificación.')
   322→        await pc.setRemoteDescription(new RTCSessionDescription(data.offer))
   323→        const answer = await pc.createAnswer()
   324→        await pc.setLocalDescription(answer)
   325→        socket.emit('webrtc-answer', { targetId: data.fromId, answer })
   326→      })
   327→
   328→      socket.on('verification-accepted', () => {
   329→        if (user) {
   330→          useDhobbytvStore.getState().setUser({ ...user, verified: true })
   331→          fetch('/api/verify-user', {
   332→            method: 'POST',
   333→            headers: { 'Content-Type': 'application/json' },
   334→            body: JSON.stringify({ username: user.username }),
   335→          })
   336→        }
   337→        toast.success('Verificado! Ya puedes usar dhobbytv')
   338→        cleanup()
   339→        useDhobbytvStore.getState().setView('main')
   340→      })
   341→
   342→      socket.on('verification-rejected', () => {
   343→        if (user) {
   344→          fetch('/api/delete-user', {
   345→            method: 'POST',
   346→            headers: { 'Content-Type': 'application/json' },
   347→            body: JSON.stringify({ username: user.username }),
   348→          })
   349→        }
   350→        toast.error('Verificación rechazada')
   351→        cleanup()
   352→        useDhobbytvStore.getState().setUser(null)
   353→        useDhobbytvStore.getState().setView('login')
   354→      })
   355→
   356→      socket.on('verification-admin-disconnected', () => {
   357→        toast.error('El administrador se desconectó')
   358→        setMessage('El admin se desconectó. Volviendo a la cola...')
   359→        setTimeout(() => {
   360→          cleanup()
   361→          useDhobbytvStore.getState().setView('verification')
   362→        }, 2000)
   363→      })
   364→    }
   365→
   366→    setup()
   367→
   368→    const cleanup = () => {
   369→      mounted = false
   370→      localStream?.getTracks().forEach((t) => t.stop())
   371→      peerRef.current?.close()
   372→      socketRef.current?.disconnect()
   373→    }
   374→
   375→    return cleanup
   376→  }, [])
   377→
   378→  return (
   379→    <div className="min-h-screen flex items-center justify-center p-4 bg-gradient-to-br from-purple-900 via-indigo-900 to-black">
   380→      <Card className="w-full max-w-lg bg-gray-900/80 border-gray-700 text-white">
   381→        <CardHeader className="text-center">
   382→          <CardTitle className="text-xl">Verificación con Administrador</CardTitle>
   383→          <p className="text-yellow-400 text-sm">Muestra tu identificación por cámara</p>
   384→        </CardHeader>
   385→        <CardContent className="space-y-4">
   386→          <div className="relative rounded-xl overflow-hidden bg-black aspect-video">
   387→            <video
   388→              ref={localVideoRef}
   389→              autoPlay
   390→              playsInline
   391→              muted
   392→              className="w-full h-full object-cover"
   393→            />
   394→            <div className="absolute bottom-2 left-2 bg-red-600 text-white text-xs px-2 py-1 rounded-full flex items-center gap-1">
   395→              <div className="w-2 h-2 bg-white rounded-full animate-pulse" />
   396→              TU CÁMARA

   406→// ==================== ADMIN PANEL ====================
   407→function AdminView() {
   408→  const user = useDhobbytvStore((s) => s.user)
   409→  const verificationQueue = useDhobbytvStore((s) => s.verificationQueue)
   410→  const setVerificationQueue = useDhobbytvStore((s) => s.setVerificationQueue)
   411→  const [activeVerification, setActiveVerification] = useState<{ socketId: string; username: string; gender: string } | null>(null)
   412→  const [stats, setStats] = useState<{ totalUsers: number; verifiedUsers: number; pendingUsers: number; bannedUsers: number; totalReports: number } | null>(null)
   413→  const socketRef = useRef<Socket | null>(null)
   414→  const remoteVideoRef = useRef<HTMLVideoElement>(null)
   415→  const peerRef = useRef<RTCPeerConnection | null>(null)
   416→  const [adminStream, setAdminStream] = useState<MediaStream | null>(null)
   417→
   418→  // Load stats
   419→  useEffect(() => {
   420→    fetch('/api/admin-stats').then((r) => r.json()).then(setStats)
   421→  }, [])
   422→
   423→  useEffect(() => {
   424→    const socket = io('/?XTransformPort=3003', { transports: ['websocket'] })
   425→    socketRef.current = socket
   426→
   427→    socket.on('connect', () => {
   428→      socket.emit('user-online', {
   429→        username: user?.username,
   430→        gender: user?.gender,
   431→        country: '',
   432→        countryCode: '',
   433→        verified: true,
   434→        isAdmin: true,
   435→      })
   436→      socket.emit('get-verification-queue')
   437→    })
   438→
   439→    socket.on('verification-queue', (data: { queue: typeof verificationQueue }) => {
   440→      setVerificationQueue(data.queue)
   441→    })
   442→
   443→    socket.on('online-count', (data: { count: number }) => {
   444→      useDhobbytvStore.getState().setOnlineCount(data.count)
   445→    })
   446→
   447→    return () => { socket.disconnect() }
   448→  }, [])
   449→
   450→  // Handle verification video
   451→  useEffect(() => {
   452→    if (!activeVerification || !socketRef.current) return
   453→
   454→    let mounted = true
   455→
   456→    const setupPeer = async () => {
   457→      // Get a dummy audio-only stream for admin (camera OFF)
   458→      const stream = new MediaStream()
   459→      const audioCtx = new AudioContext()
   460→      const oscillator = audioCtx.createOscillator()
   461→      const dest = audioCtx.createMediaStreamDestination()
   462→      oscillator.connect(dest)
   463→      oscillator.start()
   464→      stream.addTrack(dest.stream.getAudioTracks()[0])
   465→      setAdminStream(stream)
   466→
   467→      const pc = new RTCPeerConnection({
   468→        iceServers: [{ urls: 'stun:stun.l.google.com:19302' }],
   469→      })
   470→      peerRef.current = pc
   471→
   472→      stream.getTracks().forEach((track) => pc.addTrack(track, stream))
   473→
   474→      pc.onicecandidate = (e) => {
   475→        if (e.candidate) {
   476→          socketRef.current?.emit('webrtc-ice-candidate', {
   477→            targetId: activeVerification.socketId,
   478→            candidate: e.candidate,
   479→          })
   480→        }
   481→      }
   482→
   483→      pc.ontrack = (e) => {
   484→        if (remoteVideoRef.current && mounted) {
   485→          remoteVideoRef.current.srcObject = e.streams[0]
   486→        }
   487→      }
   488→
   489→      const offer = await pc.createOffer()
   490→      await pc.setLocalDescription(offer)
   491→      socketRef.current?.emit('webrtc-offer', {
   492→        targetId: activeVerification.socketId,
   493→        offer,
   494→      })
   495→    }
   496→
   497→    setupPeer()
   498→
   499→    const handleAnswer = async (data: { fromId: string; answer: RTCSessionDescriptionInit }) => {
   500→      if (peerRef.current) {
   501→        await peerRef.current.setRemoteDescription(new RTCSessionDescription(data.answer))
   502→      }
   503→    }
   504→
   505→    const handleIce = (data: { fromId: string; candidate: RTCIceCandidateInit }) => {
   506→      peerRef.current?.addIceCandidate(new RTCIceCandidate(data.candidate))
   507→    }
   508→
   509→    socketRef.current.on('webrtc-answer', handleAnswer)
   510→    socketRef.current.on('webrtc-ice-candidate', handleIce)
   511→
   512→    return () => {
   513→      mounted = false
   514→      peerRef.current?.close()
   515→      peerRef.current = null
   516→      audioCtx?.close?.()
   517→      stream?.getTracks().forEach((t) => t.stop())
   518→    }
   519→  }, [activeVerification])
   520→
   521→  const handleAccept = () => {
   522→    if (!activeVerification || !socketRef.current) return
   523→    socketRef.current.emit('admin-accept-verification', { userSocketId: activeVerification.socketId })
   524→    fetch('/api/verify-user', {
   525→      method: 'POST',
   526→      headers: { 'Content-Type': 'application/json' },
   527→      body: JSON.stringify({ username: activeVerification.username }),
   528→    })
   529→    cleanupVerification()
   530→    toast.success(`${activeVerification.username} verificado`)
   531→    fetch('/api/admin-stats').then((r) => r.json()).then(setStats)
   532→  }
   533→
   534→  const handleReject = () => {
   535→    if (!activeVerification || !socketRef.current) return
   536→    socketRef.current.emit('admin-reject-verification', { userSocketId: activeVerification.socketId })
   537→    fetch('/api/delete-user', {
   538→      method: 'POST',
   539→      headers: { 'Content-Type': 'application/json' },
   540→      body: JSON.stringify({ username: activeVerification.username }),
   541→    })
   542→    cleanupVerification()
   543→    toast.error(`${activeVerification.username} rechazado y eliminado`)
