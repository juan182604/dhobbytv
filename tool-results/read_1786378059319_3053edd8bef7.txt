    22→// ==================== LOGIN ====================
    23→function LoginView() {
    24→  const [username, setUsername] = useState('')
    25→  const [password, setPassword] = useState('')
    26→  const [loading, setLoading] = useState(false)
    27→  const setView = useDhobbytvStore((s) => s.setView)
    28→  const setUser = useDhobbytvStore((s) => s.setUser)
    29→
    30→  const handleLogin = async () => {
    31→    if (!username || !password) return toast.error('Completa todos los campos')
    32→    setLoading(true)
    33→    try {
    34→      const res = await fetch('/api/login', {
    35→        method: 'POST',
    36→        headers: { 'Content-Type': 'application/json' },
    37→        body: JSON.stringify({ username, password }),
    38→      })
    39→      const data = await res.json()
    40→      if (data.error) {
    41→        toast.error(data.error)
    42→        if (data.suspendedUntil) {
    43→          toast.info(`Hasta: ${new Date(data.suspendedUntil).toLocaleString()}`)
    44→        }
    45→        return
    46→      }
    47→      setUser(data.user)
    48→      if (data.user.isSuperAdmin) {
    49→        setView('super-admin')
    50→      } else if (data.user.isAdmin) {
    51→        setView('admin')
    52→      } else if (!data.user.verified) {
    53→        setView('verification')
    54→      } else {
    55→        setView('main')
    56→      }
    57→    } catch {
    58→      toast.error('Error de conexión')
    59→    } finally {
    60→      setLoading(false)
    61→    }
    62→  }
    63→
    64→  const announcement = useDhobbytvStore((s) => s.announcement)
    65→
    66→  return (
    67→    <div className="min-h-screen flex items-center justify-center p-4 bg-gradient-to-br from-purple-900 via-indigo-900 to-black">
    68→      {announcement && (
    69→        <div className="fixed top-0 left-0 right-0 bg-yellow-600 text-white text-center py-2 text-sm z-50">
    70→          {announcement}
    71→        </div>
    72→      )}
    73→      <Card className="w-full max-w-md bg-gray-900/80 border-gray-700 text-white">
    74→        <CardHeader className="text-center pb-2">
    75→          <CardTitle className="text-4xl font-black tracking-tight">
    76→            <span className="text-purple-400">dhobby</span>
    77→            <span className="text-green-400">tv</span>
    78→          </CardTitle>
    79→          <p className="text-gray-400 text-sm mt-1">Conecta con personas que comparten tus intereses</p>
    80→        </CardHeader>
    81→        <CardContent className="space-y-4">
    82→          <Input placeholder="Usuario" value={username} onChange={(e) => setUsername(e.target.value)} className="bg-gray-800 border-gray-600 text-white placeholder:text-gray-500" onKeyDown={(e) => e.key === 'Enter' && handleLogin()} />
    83→          <Input type="password" placeholder="Contraseña" value={password} onChange={(e) => setPassword(e.target.value)} className="bg-gray-800 border-gray-600 text-white placeholder:text-gray-500" onKeyDown={(e) => e.key === 'Enter' && handleLogin()} />
    84→          <Button onClick={handleLogin} disabled={loading} className="w-full bg-purple-600 hover:bg-purple-700 text-white font-bold py-6 text-lg">
    85→            {loading ? 'Entrando...' : 'Iniciar Sesión'}

    90→            </button>
    91→          </div>
    92→        </CardContent>

   163→// ==================== VERIFICATION WAITING ====================
   164→function VerificationView() {
   165→  const user = useDhobbytvStore((s) => s.user)
   166→  const socketRef = useRef<Socket | null>(null)
   167→
   168→  useEffect(() => {
   169→    const socket = io('/?XTransformPort=3003', { transports: ['websocket'] })
   170→    socketRef.current = socket
   171→    socket.on('connect', () => {
   172→      socket.emit('join-verification-queue', { username: user?.username, gender: user?.gender })
   173→    })
   174→    socket.on('in-verification-queue', (data: { position: number }) => {
   175→      toast.info(`Posición en la cola: ${data.position}`)
   176→    })
   177→    socket.on('start-verification', () => {
   178→      useDhobbytvStore.getState().setView('verification-video')
   179→    })
   180→    socket.on('verification-accepted', () => {
   181→      if (user) {
   182→        useDhobbytvStore.getState().setUser({ ...user, verified: true })
   183→        fetch('/api/verify-user', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ username: user.username }) })
   184→      }
   185→      toast.success('Has sido verificado!')
   186→      useDhobbytvStore.getState().setView('main')
   187→      socket.disconnect()
   188→    })
   189→    socket.on('verification-rejected', () => {
   190→      if (user) {
   191→        fetch('/api/delete-user', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ username: user.username }) })
   192→      }
   193→      toast.error('Verificación rechazada. Tu cuenta ha sido eliminada.')
   194→      useDhobbytvStore.getState().setUser(null)
   195→      useDhobbytvStore.getState().setView('login')
   196→      socket.disconnect()
   197→    })
   198→    return () => { socket.disconnect() }
   199→  }, [])
   200→
   201→  return (
   202→    <div className="min-h-screen flex items-center justify-center p-4 bg-gradient-to-br from-purple-900 via-indigo-900 to-black">
   203→      <Card className="w-full max-w-md bg-gray-900/80 border-gray-700 text-white text-center">
   204→        <CardContent className="pt-8 pb-8 space-y-4">
   205→          <div className="text-6xl animate-pulse">🔒</div>
   206→          <h2 className="text-2xl font-bold">Verificación de Edad</h2>
   207→          <p className="text-gray-400">Prepara tu documento de identificación. El admin te pedirá que lo muestres por cámara.</p>
   208→          <div className="flex items-center justify-center gap-2 text-gray-300">
   209→            <div className="w-2 h-2 bg-green-400 rounded-full animate-pulse" />
   210→            Esperando en la cola de verificación...
   211→          </div>
   212→          <Button variant="outline" className="text-gray-400 border-gray-600" onClick={() => { socketRef.current?.disconnect(); useDhobbytvStore.getState().setUser(null); useDhobbytvStore.getState().setView('login') }}>Salir</Button>
   213→        </CardContent>
   214→      </Card>
   215→    </div>
   216→  )
   217→}

   219→// ==================== VERIFICATION VIDEO ====================
   220→function VerificationVideoView() {
   221→  const user = useDhobbytvStore((s) => s.user)
   222→  const [localStream, setLocalStream] = useState<MediaStream | null>(null)
   223→  const localVideoRef = useRef<HTMLVideoElement>(null)
   224→  const peerRef = useRef<RTCPeerConnection | null>(null)
   225→  const socketRef = useRef<Socket | null>(null)
   226→  const [message, setMessage] = useState('Conectando con administrador...')
   227→
   228→  useEffect(() => {
   229→    let mounted = true
   230→    const setup = async () => {
   231→      const stream = await navigator.mediaDevices.getUserMedia({ video: true, audio: true })
   232→      if (!mounted) { stream.getTracks().forEach((t) => t.stop()); return }
   233→      setLocalStream(stream)
   234→      if (localVideoRef.current) localVideoRef.current.srcObject = stream
   235→      const socket = io('/?XTransformPort=3003', { transports: ['websocket'] })
   236→      socketRef.current = socket
   237→      const pc = new RTCPeerConnection({ iceServers: [{ urls: 'stun:stun.l.google.com:19302' }] })
   238→      peerRef.current = pc
   239→      stream.getTracks().forEach((track) => pc.addTrack(track, stream))
   240→      pc.onicecandidate = (e) => { if (e.candidate) socket.emit('webrtc-ice-candidate', { targetId: useDhobbytvStore.getState().verificationAdminSocketId, candidate: e.candidate }) }
   241→      socket.on('webrtc-offer', async (data: { fromId: string; offer: RTCSessionDescriptionInit }) => {
   242→        setMessage('Conectado con administrador. Muestra tu identificación.')
   243→        await pc.setRemoteDescription(new RTCSessionDescription(data.offer))
   244→        const answer = await pc.createAnswer()
   245→        await pc.setLocalDescription(answer)
   246→        socket.emit('webrtc-answer', { targetId: data.fromId, answer })
   247→      })
   248→      socket.on('verification-accepted', () => {
   249→        if (user) {
   250→          useDhobbytvStore.getState().setUser({ ...user, verified: true })
   251→          fetch('/api/verify-user', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ username: user.username }) })
   252→        }
   253→        toast.success('Verificado!'); cleanup(); useDhobbytvStore.getState().setView('main')
   254→      })
   255→      socket.on('verification-rejected', () => {
   256→        if (user) fetch('/api/delete-user', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ username: user.username }) })
   257→        toast.error('Rechazado'); cleanup(); useDhobbytvStore.getState().setUser(null); useDhobbytvStore.getState().setView('login')
   258→      })
   259→      socket.on('verification-admin-disconnected', () => {
   260→        toast.error('El admin se desconectó'); setMessage('Volviendo a la cola...')
   261→        setTimeout(() => { cleanup(); useDhobbytvStore.getState().setView('verification') }, 2000)
   262→      })
   263→    }
   264→    setup()
   265→    const cleanup = () => { mounted = false; localStream?.getTracks().forEach((t) => t.stop()); peerRef.current?.close(); socketRef.current?.disconnect() }
   266→    return cleanup
   267→  }, [])
   268→
   269→  return (
   270→    <div className="min-h-screen flex items-center justify-center p-4 bg-gradient-to-br from-purple-900 via-indigo-900 to-black">
   271→      <Card className="w-full max-w-lg bg-gray-900/80 border-gray-700 text-white">
   272→        <CardHeader className="text-center"><CardTitle className="text-xl">Verificación con Administrador</CardTitle><p className="text-yellow-400 text-sm">Muestra tu identificación por cámara</p></CardHeader>
   273→        <CardContent className="space-y-4">
   274→          <div className="relative rounded-xl overflow-hidden bg-black aspect-video">
   275→            <video ref={localVideoRef} autoPlay playsInline muted className="w-full h-full object-cover" />
   276→            <div className="absolute bottom-2 left-2 bg-red-600 text-white text-xs px-2 py-1 rounded-full flex items-center gap-1"><div className="w-2 h-2 bg-white rounded-full animate-pulse" />TU CÁMARA</div>
   277→          </div>
   278→          <p className="text-center text-gray-300 text-sm">{message}</p>
   279→        </CardContent>
   280→      </Card>
   281→    </div>
   282→  )
   283→}

   599→      fetch('/api/admin-stats').then((res) => res.json()),
   600→      fetch('/api/reported-users').then((res) => res.json()),
   601→    ])

   610→    socket.on('connect', () => {
   611→      socket.emit('user-online', { username: user?.username, gender: user?.gender, country: '', countryCode: '', verified: true, isAdmin: true })
   612→      socket.emit('get-verification-queue')
   613→    })

   624→      const oscillator = audioCtx.createOscillator()
   625→      const dest = audioCtx.createMediaStreamDestination()
   626→      oscillator.connect(dest); oscillator.start()

   634→      const offer = await pc.createOffer()
   635→      await pc.setLocalDescription(offer)

   640→    const handleIce = (data: { fromId: string; candidate: RTCIceCandidateInit }) => { peerRef.current?.addIceCandidate(new RTCIceCandidate(data.candidate)) }
   641→    socketRef.current.on('webrtc-answer', handleAnswer)
   642→    socketRef.current.on('webrtc-ice-candidate', handleIce)

   750→                      <div className="flex items-center gap-3">
   751→                        <span>{getGenderShort(u.gender)}</span>
   752→                        <span className="font-medium text-sm">{u.username}</span>
   753→                        <Badge variant={u._count.reports >= 5 ? 'destructive' : u._count.reports >= 3 ? 'secondary' : 'outline'} className={u._count.reports >= 3 ? 'bg-red-600 text-white' : u._count.reports >= 2 ? 'bg-orange-600 text-white' : ''}>{u._count.reports} reporte{u._count.reports !== 1 ? 's' : ''}</Badge>
   754→                        {u.banned && <Badge className="bg-red-800">BANEADO</Badge>}

   759→                        {(u.banned || (u.suspendedUntil && new Date(u.suspendedUntil) > new Date())) && <Button size="sm" className="bg-green-600 text-xs" onClick={() => { if (u.banned) handleUnban(u.id); else handleUnsuspend(u.id) }}>{u.banned ? 'Desbanear' : 'Quitar suspensión'}</Button>}
   760→                        <Button size="sm" variant="outline" className="text-xs" onClick={() => setSelectedReportedUser(selectedReportedUser?.id === u.id ? null : u)}>Ver reportes</Button>
   761→                      </div>
   762→                    </div>
   763→                    {selectedReportedUser?.id === u.id && u.reports.length > 0 && (
   764→                      <div className="mt-2 p-2 bg-gray-900 rounded-lg space-y-1">
   765→                        {u.reports.map((r: any) => (
   766→                          <div key={r.id} className="text-xs text-gray-300 flex gap-2"><span className="text-gray-500">{new Date(r.createdAt).toLocaleString()}</span><span>—</span><span>Por: {r.reporter.username}</span><span>—</span><span className="text-yellow-400">{r.reason}</span></div>
   767→                        ))}
   768→                      </div>
   769→                    )}
   770→                  </div>
   771→                ))}
   772→                {reportedUsers.length === 0 && <p className="text-gray-500 text-center py-8">No hay usuarios reportados</p>}
   773→              </div>
   774→            </ScrollArea>
   775→          </CardContent>
   776→        </Card>
   777→      </main>
   778→
   779→      {/* Ban Dialog */}
   780→      <Dialog open={!!banDialog} onOpenChange={() => setBanDialog(null)}>
   781→        <DialogContent className="bg-gray-900 border-gray-700 text-white">
   782→          <DialogHeader><DialogTitle>Banear a {banDialog?.username}</DialogTitle><DialogDescription className="text-gray-400">Baneo permanente hasta que el super admin lo revierta</DialogDescription></DialogHeader>
   783→          <Input placeholder="Razón del baneo" value={banReason} onChange={(e) => setBanReason(e.target.value)} className="bg-gray-800 border-gray-600" />
   784→          <div className="flex gap-3"><Button variant="outline" className="flex-1" onClick={() => setBanDialog(null)}>Cancelar</Button><Button className="flex-1 bg-red-600 hover:bg-red-700" onClick={handleBan}>Banear</Button></div>
   785→        </DialogContent>
   786→      </Dialog>
   787→
   788→      {/* Suspend Dialog */}
   789→      <Dialog open={!!suspendDialog} onOpenChange={() => setSuspendDialog(null)}>
   790→        <DialogContent className="bg-gray-900 border-gray-700 text-white">
   791→          <DialogHeader><DialogTitle>Suspender a {suspendDialog?.username}</DialogTitle></DialogHeader>
   792→          <Input type="number" placeholder="Horas" value={suspendHours} onChange={(e) => setSuspendHours(e.target.value)} className="bg-gray-800 border-gray-600" min="1" />
   793→          <div className="flex gap-2 flex-wrap">{[1, 6, 24, 72, 168].map((h) => (<Button key={h} size="sm" variant="outline" className={Number(suspendHours) === h ? 'bg-orange-600 border-orange-500' : ''} onClick={() => setSuspendHours(String(h))}>{h}h</Button>))}</div>
   794→          <div className="flex gap-3"><Button variant="outline" className="flex-1" onClick={() => setSuspendDialog(null)}>Cancelar</Button><Button className="flex-1 bg-orange-600 hover:bg-orange-700" onClick={handleSuspend}>Suspender</Button></div>
   795→        </DialogContent>
   796→      </Dialog>
   797→    </div>
   798→  )

   845→    }

   848→    return pc

   869→      if (peerRef.current) { await peerRef.current.setRemoteDescription(new RTCSessionDescription(data.offer)); const answer = await peerRef.current.createAnswer(); await peerRef.current.setLocalDescription(answer); socket.emit('webrtc-answer', { targetId: data.fromId, answer }) }
   870→    })
